cmake_minimum_required(VERSION 3.25)
project(SpiteTooling LANGUAGES CXX)
find_package(Spite CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_path(STB_INCLUDE_DIR stb_image_write.h REQUIRED)

set(modules Animation AnimationAuthoring Application Audio Asset AssetAuthoring AssetRuntime Base Control Ecs Engine EngineTools Localization Physics Profiling Rendering RenderObservation RenderPipeline RenderRuntime Scene Serialization Ui UiRendering UiView World)
foreach(module IN LISTS modules)
    list(APPEND sdk_libraries Spite::${module})
endforeach()
set(editor_root "${CMAKE_CURRENT_LIST_DIR}/../source/open/editor")
file(GLOB editor_sources CONFIGURE_DEPENDS "${editor_root}/*.cpp")
if(NOT editor_sources)
    message(FATAL_ERROR "The SDK package must include editor sources in source/open/editor")
endif()
add_library(SpiteEditorSource SHARED ${editor_sources})
target_include_directories(SpiteEditorSource PRIVATE "${editor_root}" "${STB_INCLUDE_DIR}")
target_link_libraries(SpiteEditorSource PRIVATE ${sdk_libraries} nlohmann_json::nlohmann_json)
target_compile_definitions(SpiteEditorSource PRIVATE SPITE_EDITOR_BUILD_DLL)
spite_configure_module(SpiteEditorSource)
target_precompile_headers(SpiteEditorSource PRIVATE "${CMAKE_CURRENT_LIST_DIR}/ToolingPch.hpp")
set_target_properties(SpiteEditorSource PROPERTIES OUTPUT_NAME SpiteEditor)
install(TARGETS SpiteEditorSource RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)

# Use the SDK's stock host with a DLL built against the same SDK and compatible ABI.
option(SPITE_BUILD_SOURCE_CLI "Build the packaged open CLI sources" OFF)
if(SPITE_BUILD_SOURCE_CLI)
    if(NOT WIN32)
        message(FATAL_ERROR "The source CLI is supported on Windows only")
    endif()
    set(cli_source "${CMAKE_CURRENT_LIST_DIR}/../tools/SpiteCli/Main.cpp")
    if(NOT EXISTS "${cli_source}")
        message(FATAL_ERROR "Source CLI requested, but tools/SpiteCli/Main.cpp is not packaged; the SDK publisher must stage its complete open build inputs")
    endif()
    find_package(SQLite3 REQUIRED)
    add_executable(spite-source "${cli_source}")
    target_link_libraries(spite-source PRIVATE Spite::Base nlohmann_json::nlohmann_json SQLite::SQLite3)
    spite_configure_module(spite-source)
    target_precompile_headers(spite-source PRIVATE "${CMAKE_CURRENT_LIST_DIR}/ToolingPch.hpp")
    set_target_properties(spite-source PROPERTIES OUTPUT_NAME spite)
    install(TARGETS spite-source RUNTIME DESTINATION bin)
endif()
if(WIN32)
    target_compile_definitions(SpiteEditorSource PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOGDI)
    if(TARGET spite-source)
        target_compile_definitions(spite-source PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOGDI)
    endif()
endif()
message(STATUS "Use the SDK's stock host with the rebuilt SpiteEditor library from the same SDK and compatible toolchain/ABI")
