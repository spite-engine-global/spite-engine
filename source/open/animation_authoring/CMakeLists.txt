cmake_minimum_required(VERSION 3.25)
project(SpiteAnimationAuthoring LANGUAGES CXX)
find_package(Spite CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(SpiteAnimationAuthoring SHARED AnimationController.cpp)
target_compile_features(SpiteAnimationAuthoring PRIVATE cxx_std_23)
target_compile_definitions(SpiteAnimationAuthoring PRIVATE SPITE_ANIMATION_AUTHORING_BUILD_DLL)
target_link_libraries(SpiteAnimationAuthoring PRIVATE Spite::Base nlohmann_json::nlohmann_json)
spite_configure_module(SpiteAnimationAuthoring)
